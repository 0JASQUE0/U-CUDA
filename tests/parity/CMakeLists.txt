# Forward public module include names to the actual upstream headers. No compute
# source or upstream header is rewritten, copied from the modular implementation,
# or replaced with a stub.
file(GLOB_RECURSE module_headers "${parity_root}/libs/*/include/ucuda/*.h"
    "${parity_root}/libs/*/include/ucuda/*.hpp" "${parity_root}/libs/*/include/ucuda/*.cuh")
foreach(header IN LISTS module_headers)
    string(REGEX REPLACE ".*/include/" "" relative "${header}")
    get_filename_component(name "${header}" NAME)
    set(original "${name}")
    if(name STREQUAL "runtime_settings.h" OR name STREQUAL "analysis_types.h")
        set(original parametric_engine.h)
    elseif(name STREQUAL "system.h")
        set(original codegen.hpp)
    elseif(name STREQUAL "plot_window.h")
        set(original app_model.h)
    elseif(name STREQUAL "export_snapshot.h")
        set(original data_export.h)
    endif()
    if(EXISTS "${upstream}/${original}")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/forward/${relative}" "#pragma once\n#include \"${upstream}/${original}\"\n")
    endif()
endforeach()
configure_file(platform_paths.h "forward/ucuda/platform/platform_paths.h" COPYONLY)

function(reference_executable target)
    add_executable(${target} ${ARGN} "${parity_root}/tests/utf8.manifest")
    target_link_libraries(${target} PRIVATE upstream_original)
    target_compile_definitions(${target} PRIVATE UCUDA_TEST_SOURCE="${parity_root}")
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND}
        -Dsource=${upstream} -Ddestination=$<TARGET_FILE_DIR:${target}>
        -P "${CMAKE_CURRENT_SOURCE_DIR}/PackOriginal.cmake")
    if(NOT target STREQUAL "main_benchmark")
        add_test(NAME ${target} COMMAND ${target})
        set_tests_properties(${target} PROPERTIES RUN_SERIAL TRUE TIMEOUT 3600)
    endif()
endfunction()
