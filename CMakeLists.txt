cmake_minimum_required(VERSION 3.25)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
endif()
project(U_CUDA VERSION 0.1.0 LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
option(U_CUDA_BUILD_GUI "Build the desktop GUI" ON)
option(U_CUDA_BUILD_LEGACY "Build the legacy batch calculation executable" OFF)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
include(cmake/Targets.cmake)
include(cmake/Runtime.cmake)
add_subdirectory(libs/domain)
add_subdirectory(libs/platform)
add_subdirectory(libs/equations)
add_subdirectory(libs/numerics)
add_subdirectory(libs/storage)
add_subdirectory(libs/cuda)
add_subdirectory(libs/analysis)
add_subdirectory(libs/ocr)
add_subdirectory(libs/application)
if(U_CUDA_BUILD_GUI)
    include(FetchContent)
    add_subdirectory(third_party)
    add_subdirectory(libs/plotting)
    add_subdirectory(libs/ui)
    add_subdirectory(apps/desktop)
endif()
if(U_CUDA_BUILD_LEGACY)
    add_subdirectory(apps/legacy)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
